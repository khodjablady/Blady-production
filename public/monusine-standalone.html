Architecture C# (.NET 8/9 Monolithe Modulaire)

Exemple de code C# backend minimal API et service de calcul TRS réutilisables dans une solution .NET standard.

BladyProduction.Mes.Services/OeeCalculatorService.cs C# 12 / .NET 8-9 
namespace BladyProduction.Mes.Services;

// Record C# immuable pour les métriques TRS
public record OeeMetricsDto(
    double Disponibilite, 
    double Performance, 
    double Qualite, 
    double TrsGlobal
);

public class OeeCalculatorService
{
    public OeeMetricsDto CalculerTrs(double tempsFonctionnementMin, double tempsOuvertureMin, 
                                     int piecesProduites, int piecesTheoriques, int rebuts)
    {
        var d = Math.Clamp(tempsFonctionnementMin / tempsOuvertureMin * 100.0, 0, 100);
        var p = Math.Clamp((double)piecesProduites / piecesTheoriques * 100.0, 0, 100);
        var q = piecesProduites > 0 ? Math.Clamp((double)(piecesProduites - rebuts) / piecesProduites * 100.0, 0, 100) : 100;
        var trs = Math.Round((d / 100.0) * (p / 100.0) * (q / 100.0) * 100.0, 1);

        return new OeeMetricsDto(Math.Round(d, 1), Math.Round(p, 1), Math.Round(q, 1), trs);
    }
}
